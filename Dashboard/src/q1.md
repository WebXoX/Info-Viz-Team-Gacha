# Q1: Airport and Carrier Rankings

**Research question:** Which US domestic airports move the most passengers, freight, mail, and estimated payload, and how do their rankings differ?

This page compares **departing airport traffic**. Rankings are calculated nationwide for each year, not just among the airports selected below.

```js
import * as d3 from "npm:d3";

const airports = await FileAttachment("./data/q1_airports_year_all_classes.csv")
  .csv({typed: true});

const measures = [
  {rank: "passengers_rank", value: "passengers", label: "Passengers", unit: "segments"},
  {rank: "freight_lb_rank", value: "freight_lb", label: "Freight", unit: "lb"},
  {rank: "mail_lb_rank", value: "mail_lb", label: "Mail", unit: "lb"},
  {rank: "estimated_payload_carried_lb_rank", value: "estimated_payload_carried_lb", label: "Estimated payload", unit: "lb"}
];

const airportInfo = Array.from(
  d3.group(airports.filter(d => d.airport), d => d.airport),
  ([code, records]) => ({
    code,
    city: records.find(d => d.city)?.city ?? "",
    state: records.find(d => d.state)?.state ?? ""
  })
).sort((a, b) => d3.ascending(a.code, b.code));

const validCodes = new Set(airportInfo.map(d => d.code));
const initialCodes = ["ATL", "LAX", "MEM"].filter(d => validCodes.has(d));
const selection = Mutable(initialCodes);

function addAirport(code) {
  if (!validCodes.has(code) || selection.value.includes(code) || selection.value.length >= 5) return;
  selection.value = [...selection.value, code];
}

function removeAirport(code) {
  selection.value = selection.value.filter(d => d !== code);
}
```

## Top five airports by measurement

These rankings update when you change the year. Click an airport to add it to the comparison (maximum five).

```js
{
  const grid = document.createElement("div");
  grid.style.cssText = "display:grid;grid-template-columns:repeat(auto-fit,minmax(210px,1fr));gap:12px;margin:12px 0 24px";

  for (const m of measures) {
    const card = document.createElement("section");
    card.style.cssText = "border:1px solid #cbd5e1;border-radius:10px;padding:14px;min-width:0";
    const heading = document.createElement("h3");
    heading.textContent = m.label;
    heading.style.cssText = "margin:0 0 12px;font-size:16px";
    card.append(heading);

    const top5 = yearData
      .filter(d => Number.isFinite(d[m.rank]) && d[m.rank] > 0 && d[m.value] > 0)
      .slice()
      .sort((a, b) => d3.ascending(a[m.rank], b[m.rank]) || d3.ascending(a.airport, b.airport))
      .slice(0, 5);

    for (const d of top5) {
      const button = document.createElement("button");
      button.type = "button";
      button.textContent = `#${d[m.rank]}  ${d.airport}  ·  ${d3.format(",.0f")(d[m.value])}`;
      button.title = `Add ${d.airport} to the comparison`;
      button.disabled = selection.includes(d.airport) || selection.length >= 5;
      button.style.cssText = "display:block;width:100%;margin:5px 0;padding:9px;text-align:left;border:1px solid #cbd5e1;border-radius:6px;cursor:pointer;background:transparent;color:inherit";
      button.addEventListener("click", () => addAirport(d.airport));
      card.append(button);
    }
    if (!top5.length) {
      const empty = document.createElement("p");
      empty.textContent = "No ranking data for this year.";
      card.append(empty);
    }
    grid.append(card);
  }
  display(grid);
}
```

## Compare airports

Search by airport code or city, then click **Add**. Remove an airport using its tag below. You can compare up to five airports.

```js
{
  const root = document.createElement("div");
  root.style.cssText = "max-width:540px;margin:12px 0";
  const input = document.createElement("input");
  input.type = "search";
  input.placeholder = "Search airports, e.g. JFK or Atlanta";
  input.setAttribute("aria-label", "Search airports");
  input.style.cssText = "box-sizing:border-box;width:100%;padding:11px;border:1px solid #94a3b8;border-radius:8px;font:inherit";
  const results = document.createElement("div");
  results.style.cssText = "margin-top:8px;display:grid;gap:5px";

  function renderResults() {
    results.replaceChildren();
    const query = input.value.trim().toLowerCase();
    if (!query) return;
    const found = airportInfo.filter(a =>
      !selection.includes(a.code) &&
      `${a.code} ${a.city} ${a.state}`.toLowerCase().includes(query)
    ).slice(0, 8);

    for (const airport of found) {
      const button = document.createElement("button");
      button.type = "button";
      button.textContent = `+ ${airport.code} · ${airport.city}${airport.state ? `, ${airport.state}` : ""}`;
      button.disabled = selection.length >= 5;
      button.style.cssText = "text-align:left;padding:10px;border:1px solid #cbd5e1;border-radius:6px;background:transparent;color:inherit;cursor:pointer";
      button.addEventListener("click", () => {
        addAirport(airport.code);
        input.value = "";
        renderResults();
      });
      results.append(button);
    }
    if (!found.length) {
      const message = document.createElement("p");
      message.textContent = "No matching unselected airports.";
      results.append(message);
    }
  }

  input.addEventListener("input", renderResults);
  root.append(input, results);
  display(root);
}
```

## Airport ranking bump chart

The horizontal positions represent different measures, **not years**. Rank 1 is at the top. Hover over a point to see a detailed tooltip with the airport, year, national rank, and actual value.

```js
{
  const selectedData = yearData.filter(d => selection.includes(d.airport));
  const w = 860;
  const h = 480;
  const margin = {top: 55, right: 55, bottom: 88, left: 85};

  const x = d3.scalePoint()
    .domain(measures.map(m => m.label))
    .range([margin.left, w - margin.right]);

  const ranks = selectedData.flatMap(d => measures
    .map(m => d[m.rank])
    .filter(v => Number.isFinite(v) && v > 0));
  const maxRank = Math.max(10, d3.max(ranks) ?? 10);
  const y = d3.scaleLinear()
    .domain([1, maxRank])
    .range([margin.top, h - margin.bottom]);

  // Stable colors for currently selected airports.
  const color = d3.scaleOrdinal()
    .domain(selection)
    .range(d3.schemeTableau10);

  // Tooltip stays inside this chart container and follows the pointer.
  const chart = document.createElement("div");
  chart.style.cssText = "position:relative;width:100%";

  const tooltip = document.createElement("div");
  tooltip.setAttribute("role", "tooltip");
  tooltip.style.cssText = `
    position:absolute;
    z-index:10;
    display:none;
    pointer-events:none;
    max-width:250px;
    padding:10px 13px;
    border:1px solid #cbd5e1;
    border-radius:8px;
    background:#fff;
    color:#172033;
    box-shadow:0 4px 16px rgba(0,0,0,.15);
    font:13px/1.55 system-ui,sans-serif;
  `;

  function moveTooltip(event) {
    const bounds = chart.getBoundingClientRect();
    const left = event.clientX - bounds.left;
    const top = event.clientY - bounds.top;
    tooltip.style.left = `${Math.max(4, Math.min(left + 14, bounds.width - tooltip.offsetWidth - 4))}px`;
    tooltip.style.top = `${Math.max(4, top - tooltip.offsetHeight - 14)}px`;
  }

  function showTooltip(event, airport, point) {
    tooltip.replaceChildren();
    const title = document.createElement("strong");
    title.textContent = `${airport.airport} · ${airport.city || "Airport"}`;
    title.style.display = "block";
    title.style.marginBottom = "5px";
    tooltip.append(title);

    for (const text of [
      `Year: ${year}`,
      `Measure: ${point.label}`,
      `National rank: #${point.rank}`,
      `Value: ${d3.format(",.0f")(point.value)} ${point.unit}`
    ]) {
      const row = document.createElement("div");
      row.textContent = text;
      tooltip.append(row);
    }
    tooltip.style.display = "block";
    moveTooltip(event);
  }

  const svg = d3.create("svg")
    .attr("viewBox", `0 0 ${w} ${h}`)
    .attr("role", "img")
    .attr("aria-label", `National airport ranking comparison for ${year}`)
    .style("max-width", "100%")
    .style("height", "auto");

  svg.append("g")
    .attr("transform", `translate(0,${h - margin.bottom})`)
    .call(d3.axisBottom(x));

  svg.append("g")
    .attr("transform", `translate(${margin.left},0)`)
    .call(d3.axisLeft(y).ticks(Math.min(12, maxRank)).tickFormat(d3.format("d")));

  svg.append("text")
    .attr("x", margin.left)
    .attr("y", 24)
    .attr("font-size", 14)
    .attr("fill", "currentColor")
    .text(`Nationwide airport ranks · ${year} (1 = highest)`);

  const line = d3.line()
    .defined(p => Number.isFinite(p.rank) && p.rank > 0)
    .x(p => x(p.label))
    .y(p => y(p.rank));

  for (const airport of selectedData) {
    const points = measures.map(m => ({
      label: m.label,
      rank: airport[m.rank],
      value: airport[m.value],
      unit: m.unit
    }));

    svg.append("path")
      .datum(points)
      .attr("d", line)
      .attr("fill", "none")
      .attr("stroke", color(airport.airport))
      .attr("stroke-width", 3);

    svg.append("g")
      .selectAll("circle")
      .data(points.filter(p => Number.isFinite(p.rank) && p.rank > 0))
      .join("circle")
      .attr("cx", p => x(p.label))
      .attr("cy", p => y(p.rank))
      .attr("r", 6)
      .attr("fill", color(airport.airport))
      .attr("stroke", "white")
      .attr("stroke-width", 1.5)
      .style("cursor", "crosshair")
      .on("pointerenter", (event, point) => {
        d3.select(event.currentTarget).attr("r", 9);
        showTooltip(event, airport, point);
      })
      .on("pointermove", moveTooltip)
      .on("pointerleave", event => {
        d3.select(event.currentTarget).attr("r", 6);
        tooltip.style.display = "none";
      });
  }

  const legend = svg.append("g")
    .attr("transform", `translate(${margin.left},${h - 28})`);
  selection.forEach((code, i) => {
    const item = legend.append("g")
      .attr("transform", `translate(${i * 140},0)`);
    item.append("circle").attr("r", 5).attr("fill", color(code));
    item.append("text").attr("x", 11).attr("y", 4)
      .attr("font-size", 12).attr("fill", "currentColor")
      .text(code);
  });

  if (!selectedData.length) {
    svg.append("text")
      .attr("x", w / 2).attr("y", h / 2)
      .attr("text-anchor", "middle")
      .attr("fill", "currentColor")
      .text("Select airports above to display their rankings.");
  }

  chart.append(svg.node(), tooltip);
  display(chart);
}
```

## What does this visualization show?

The bump chart compares the **national rankings** of selected US domestic departure airports across passengers, freight, mail, and estimated payload carried. Each airport is ranked against the entire national dataset for the selected year, even when only a few airports are displayed.

Ranks tied on a measure share the same position. Zero measurements are unranked. Estimated payload is an approximation: **200 lb per passenger including baggage**, plus freight and mail. It is **not** available aircraft payload capacity.
