# Q2: Aviation Recovery (2015–2025)

How have US domestic passenger traffic, seating capacity, and flight departures changed nationally, across states and airports, relative to 2019?

We use scheduled passenger service (Class F), count **departing passenger segments**, and index **each measure against its own 2019 total (2019 = 100)**. For example, 110 means 10% above its 2019 level.

```js
import * as d3 from "npm:d3";

const nationalRaw = await FileAttachment("./data/q2_national_year_F.csv").csv({typed: true});
const airportRaw = await FileAttachment("./data/q2_airport_year_F.csv").csv({typed: true});

const metrics = [
  {key: "passengers", label: "Passengers", color: "#2563eb", unit: "passenger segments"},
  {key: "seats", label: "Available seats", color: "#ea580c", unit: "seats"},
  {key: "departures_performed", label: "Performed departures", color: "#059669", unit: "departures"}
];

const fmt = d3.format(",.0f");
const fmtIndex = d3.format(".1f");
const validNumber = value => value != null && value !== "" && Number.isFinite(Number(value));
const num = value => validNumber(value) ? Number(value) : null;
const total = (rows, key) => {
  const known = rows.map(d => num(d[key])).filter(v => v !== null);
  return known.length ? d3.sum(known) : null;
};

// Keep undefined baselines and unknown totals as missing (not zero).
function indexBy2019(rows, key, groupField = null) {
  const baselines = new Map(rows.filter(d => +d.year === 2019)
    .map(d => [groupField ? d[groupField] : "national", num(d[key])]));
  return rows.map(d => {
    const baseline = baselines.get(groupField ? d[groupField] : "national");
    const value = num(d[key]);
    return {...d, index: baseline != null && baseline > 0 && value != null
      ? 100 * value / baseline : null};
  });
}

// Shared tooltip for the D3 SVG charts, kept inside a positioned wrapper.
function createChartFrame() {
  const frame = document.createElement("div");
  frame.style.position = "relative";
  frame.style.width = "100%";
  const tooltip = document.createElement("div");
  Object.assign(tooltip.style, {
    position: "absolute", pointerEvents: "none", display: "none",
    background: "#111827", color: "white", borderRadius: "8px",
    padding: "10px 12px", font: "12px/1.6 system-ui, sans-serif",
    boxShadow: "0 6px 18px #0003", zIndex: "5", maxWidth: "230px"
  });
  frame.append(tooltip);
  function show(event, lines) {
    tooltip.replaceChildren();
    for (const [text, color, isTitle] of lines) {
      const row = document.createElement("div");
      row.textContent = text;
      if (color) row.style.color = color;
      if (isTitle) row.style.fontWeight = "700";
      tooltip.append(row);
    }
    tooltip.style.display = "block";
    move(event);
  }
  function move(event) {
    const rect = frame.getBoundingClientRect();
    const box = tooltip.getBoundingClientRect();
    const left = event.clientX - rect.left + 12;
    const top = event.clientY - rect.top + 12;
    tooltip.style.left = `${Math.max(0, Math.min(left, rect.width - box.width))}px`;
    tooltip.style.top = `${Math.max(0, top + box.height > rect.height ? top - box.height - 22 : top)}px`;
  }
  const hide = () => { tooltip.style.display = "none"; };
  return {frame, show, move, hide};
}

function lineChart(series, {width = 900, height = 405, yLabel = "Recovery index (2019 = 100)", valueUnit = null} = {}) {
  const margin = {top: 42, right: 190, bottom: 62, left: 80};
  const all = series.flatMap(s => s.values).filter(d => d.index != null && Number.isFinite(d.index));
  if (!all.length) {
    const p = document.createElement("p");
    p.textContent = "No valid 2019 baseline was found for this selection.";
    return p;
  }

  const {frame, show, move, hide} = createChartFrame();
  const x = d3.scaleLinear().domain([2015, 2025]).range([margin.left, width - margin.right]);
  const yMax = Math.max(120, d3.max(all, d => d.index) * 1.08);
  const y = d3.scaleLinear().domain([0, yMax]).range([height-margin.bottom, margin.top]);
  const svg = d3.create("svg")
    .attr("viewBox", [0, 0, width, height])
    .style("max-width", "100%")
    .style("height", "auto");

  svg.append("line")
    .attr("x1", margin.left).attr("x2", width-margin.right)
    .attr("y1", y(100)).attr("y2", y(100))
    .attr("stroke", "#6b7280").attr("stroke-dasharray", "5,4");
  svg.append("text")
    .attr("x", width-margin.right-5).attr("y", y(100)-8)
    .attr("text-anchor", "end").attr("font-size", 11)
    .attr("fill", "#6b7280").text("2019 baseline = 100");

  svg.append("g")
    .attr("transform", `translate(0,${height-margin.bottom})`)
    .call(d3.axisBottom(x).tickValues(d3.range(2015, 2026)).tickFormat(d3.format("d")))
    .call(g => g.selectAll("text").attr("fill", "currentColor"));
  svg.append("g")
    .attr("transform", `translate(${margin.left},0)`)
    .call(d3.axisLeft(y).ticks(6))
    .call(g => g.selectAll("text").attr("fill", "currentColor"));

  svg.append("text")
    .attr("x", (margin.left + width-margin.right)/2)
    .attr("y", height-12).attr("text-anchor", "middle")
    .attr("font-size", 13).attr("font-weight", 600)
    .attr("fill", "currentColor").text("Year");
  svg.append("text")
    .attr("transform", `translate(20,${(margin.top + height-margin.bottom)/2}) rotate(-90)`)
    .attr("text-anchor", "middle")
    .attr("font-size", 13).attr("font-weight", 600)
    .attr("fill", "currentColor").text(yLabel);

  const line = d3.line()
    .defined(d => d.index != null && Number.isFinite(d.index))
    .x(d => x(+d.year)).y(d => y(d.index));

  series.forEach((s, i) => {
    const dash = [null, "7,4", "2,4", "9,3,2,3", "4,2"][i % 5];
    const path = svg.append("path")
      .datum(s.values).attr("fill", "none")
      .attr("stroke", s.color).attr("stroke-width", 2.6)
      .attr("d", line);
    if (dash) path.attr("stroke-dasharray", dash);

    const circles = svg.append("g").selectAll("circle")
      .data(s.values.filter(d => d.index != null && Number.isFinite(d.index)))
      .join("circle")
      .attr("cx", d => x(+d.year))
      .attr("cy", d => y(d.index))
      .attr("r", 4.5).attr("fill", s.color)
      .attr("stroke", "white").attr("stroke-width", 1)
      .style("cursor", "crosshair");
    circles.on("mouseenter", function(event, d) {
      d3.select(this).attr("r", 7);
      path.attr("stroke-width", 4);
      const unit = s.unit ?? valueUnit ?? "";
      show(event, [
        [s.name, s.color, true],
        [`Year: ${d.year}`],
        [`2019 index: ${fmtIndex(d.index)}`],
        [`Actual: ${d.raw == null ? "Unavailable" : fmt(d.raw)}${unit ? " " + unit : ""}`]
      ]);
    }).on("mousemove", move)
      .on("mouseleave", function() {
        d3.select(this).attr("r", 4.5);
        path.attr("stroke-width", 2.6);
        hide();
      });

    // Legend text shares the exact series color, plus a matching line pattern.
    const legend = svg.append("g")
      .attr("transform", `translate(${width-margin.right+18},${margin.top+i*35})`);
    const swatch = legend.append("line")
      .attr("x1", 0).attr("x2", 28)
      .attr("y1", 0).attr("y2", 0)
      .attr("stroke", s.color).attr("stroke-width", 3);
    if (dash) swatch.attr("stroke-dasharray", dash);
    legend.append("text")
      .attr("x", 36).attr("y", 4)
      .attr("fill", s.color).attr("font-weight", 600)
      .attr("font-size", 12).text(s.name);
  });

  frame.prepend(svg.node());
  return frame;
}
```

## 1. National recovery

The three lines compare passengers, available seats, and performed departures using **each measure's own 2019 baseline**, avoiding incompatible raw units.

```js
const national = [...nationalRaw].sort((a,b) => +a.year - +b.year);
const nationalSeries = metrics.map(m => ({
  name: m.label,
  color: m.color,
  unit: m.unit,
  values: indexBy2019(national, m.key).map(d => ({
    year: +d.year, index: d.index, raw: num(d[m.key])
  }))
}));
display(lineChart(nationalSeries));
```

## 2. State or airport recovery comparison

Choose a geographic level, metric, and up to five locations. Each selected location is compared to **its own** 2019 value.

```js
const geoLevel = view(Inputs.select(["State", "Airport"], {label: "Geographic level", value: "State"}));
const comparisonMetric = view(Inputs.select(metrics.map(m => m.key), {
  label: "Measure", value: "passengers",
  format: key => metrics.find(m => m.key === key)?.label ?? key
}));
```

```js
// State groups are based on the origin airport's state.
const geoField = geoLevel === "State" ? "state" : "airport";
const geoRows = d3.rollups(
  airportRaw.filter(d => d[geoField] && +d.year >= 2015 && +d.year <= 2025),
  rows => Object.fromEntries(metrics.map(m => [m.key, total(rows, m.key)])),
  d => +d.year,
  d => String(d[geoField])
).flatMap(([year, groups]) => groups.map(([location, totals]) => ({year, location, ...totals})));

const geoOptions = [...new Set(geoRows.map(d => d.location))].sort();
const preferred = geoLevel === "State" ? ["CA", "FL", "TX", "GA", "IL"] : ["ATL", "DFW", "DEN", "JFK", "LAX"];
const geoDefaults = preferred.filter(x => geoOptions.includes(x));
const geoSelection = view(Inputs.select(geoOptions, {
  label: "Select up to five locations (Cmd/Ctrl-click to select multiple)",
  multiple: true, value: geoDefaults, size: 7
}));
```

```js
const chosen = [...geoSelection].slice(0, 5);
const palette = ["#2563eb", "#ea580c", "#059669", "#7c3aed", "#db2777"];
const recoverySeries = chosen.map((location, i) => {
  const rows = geoRows.filter(d => d.location === location).sort((a,b) => a.year - b.year);
  return {
    name: location,
    color: palette[i],
    unit: metrics.find(m => m.key === comparisonMetric)?.unit ?? "",
    values: indexBy2019(rows, comparisonMetric).map(d => ({
      year: d.year, index: d.index, raw: num(d[comparisonMetric])
    }))
  };
});
display(lineChart(recoverySeries));
```

## 3. Contribution to national passenger traffic

This chart shows the percentage of national **departing passenger segments** contributed by each location in a selected year. It shows size, **not** recovery.

```js
const shareLevel = view(Inputs.select(["State", "Airport"], {
  label: "Contribution level", value: "State"
}));
const shareYear = view(Inputs.range([2015, 2025], {label: "Year", value: 2025, step: 1}));
const topN = view(Inputs.range([5, 20], {label: "Show top N", value: 10, step: 1}));
```

```js
const shareField = shareLevel === "State" ? "state" : "airport";
const shareAirports = airportRaw.filter(d => +d.year === shareYear && d[shareField]);
const nationalPassengers = num(national.find(d => +d.year === shareYear)?.passengers);
const shareRows = d3.rollups(shareAirports,
  rows => total(rows, "passengers"),
  d => String(d[shareField])
).map(([location, passengers]) => ({
  location, passengers,
  share: nationalPassengers != null && nationalPassengers > 0 && passengers != null
    ? 100 * passengers / nationalPassengers : null
})).filter(d => d.share != null && d.passengers > 0)
  .sort((a,b) => d3.descending(a.passengers, b.passengers))
  .slice(0, topN);

const barWidth = 860;
const barHeight = Math.max(255, shareRows.length * 34 + 90);
const barMargin = {top: 25, right: 100, bottom: 60, left: 95};
const bx = d3.scaleLinear()
  .domain([0, d3.max(shareRows, d => d.share) || 1]).nice()
  .range([barMargin.left, barWidth-barMargin.right]);
const by = d3.scaleBand()
  .domain(shareRows.map(d => d.location))
  .range([barMargin.top, barHeight-barMargin.bottom]).padding(0.22);

const {frame: shareFrame, show: showShare, move: moveShare, hide: hideShare} = createChartFrame();
const barSvg = d3.create("svg")
  .attr("viewBox", [0,0,barWidth,barHeight])
  .style("max-width", "100%").style("height", "auto");

const bars = barSvg.append("g").selectAll("rect")
  .data(shareRows).join("rect")
  .attr("x", barMargin.left).attr("y", d => by(d.location))
  .attr("width", d => bx(d.share)-barMargin.left)
  .attr("height", by.bandwidth()).attr("fill", "#2563eb")
  .style("cursor", "crosshair");

bars.on("mouseenter", function(event,d) {
    d3.select(this).attr("fill", "#1d4ed8");
    showShare(event, [
      [d.location, "#93c5fd", true],
      [`Year: ${shareYear}`],
      [`Passenger segments: ${fmt(d.passengers)}`],
      [`National share: ${fmtIndex(d.share)}%`]
    ]);
  }).on("mousemove", moveShare)
  .on("mouseleave", function() {
    d3.select(this).attr("fill", "#2563eb");
    hideShare();
  });

barSvg.append("g").attr("transform", `translate(${barMargin.left},0)`)
  .call(d3.axisLeft(by))
  .call(g => g.selectAll("text").attr("fill", "#2563eb").attr("font-weight", 600));
barSvg.append("g").attr("transform", `translate(0,${barHeight-barMargin.bottom})`)
  .call(d3.axisBottom(bx).ticks(6).tickFormat(d => d + "%"))
  .call(g => g.selectAll("text").attr("fill", "currentColor"));
barSvg.append("text")
  .attr("x", (barMargin.left + barWidth-barMargin.right)/2)
  .attr("y", barHeight-12).attr("text-anchor", "middle")
  .attr("font-size", 13).attr("font-weight", 600)
  .attr("fill", "currentColor").text("Share of national passenger segments (%)");
barSvg.append("text")
  .attr("transform", `translate(20,${(barMargin.top + barHeight-barMargin.bottom)/2}) rotate(-90)`)
  .attr("text-anchor", "middle").attr("font-size", 13).attr("font-weight", 600)
  .attr("fill", "currentColor")
  .text(shareLevel === "State" ? "Origin state" : "Origin airport");
barSvg.append("g").selectAll("text")
  .data(shareRows).join("text")
  .attr("x", d => bx(d.share)+6)
  .attr("y", d => by(d.location)+by.bandwidth()/2)
  .attr("dominant-baseline", "middle")
  .attr("font-size", 11).attr("fill", "#2563eb")
  .text(d => `${fmtIndex(d.share)}%`);

shareFrame.prepend(barSvg.node());
display(shareFrame);
```

## What the visualizations show

- **National recovery:** Passengers, seats, and departures each use their own 2019 baseline. Their legend text, lines, and markers share the same color, with line patterns offering an additional distinction.
- **State or airport recovery:** Compare selected places on a consistent normalized scale. Missing or zero 2019 baselines are not assigned recovery indices.
- **National share:** Compare the contribution of different states or airports to national departing passenger segments for the selected year. This is not a recovery index.

**Limitations:** Passenger counts represent departures, not unique travelers. State totals are attributed to the origin airport, not to passenger residence. National totals may include territories depending on the dataset scope. The charts are descriptive and do not explain causes of recovery.
